```bash
write_ci() {
    mkdir -p .github/workflows
    local lang; lang="$(json_get .keel/profile.json stack.language || true)"
    local test_cmd; test_cmd="$(json_get .keel/profile.json verify.test || true)"
    {
      printf 'name: CI\n\non:\n  push:\n    branches: [main]\n  pull_request:\n\njobs:\n'
      printf '  verify:\n    runs-on: ubuntu-latest\n    steps:\n      - uses: actions/checkout@v4\n'
      case "$lang" in
        javascript|typescript) printf '      - uses: actions/setup-node@v4\n        with:\n          node-version: "20"\n' ;;
        python)                printf '      - uses: actions/setup-python@v5\n        with:\n          python-version: "3.11"\n' ;;
        go)                    printf '      - uses: actions/setup-go@v5\n        with:\n          go-version: "1.22"\n' ;;
      esac
      # No continue-on-error. A test step that cannot fail the build is documentation.
      printf '\n      - name: Test\n        run: %s\n' "${test_cmd:-echo 'no test command in .keel/profile.json'}"
      # Every other verify command that is a real, runnable string. test_one is templated with
      # {path} or {name} and there is nothing in CI to fill the template with, and format_fix
      # rewrites files rather than checking them: both excluded on the grounds doctor already uses.
      local k c
      for k in test_integration lint format typecheck build e2e security; do
          c="$(json_get .keel/profile.json "verify.$k" || true)"
          [ -n "$c" ] || continue
          printf '\n      - name: %s\n        run: %s\n' "$k" "$c"
      done
      printf '\n      - name: Refuse committed key material\n        run: |\n'
      printf '          if git ls-files | grep -qE "\\.(pem|key|pfx|p12|jks)$|^\\.env$"; then\n'
      printf '            echo "::error::key material is tracked in git"; git ls-files | grep -E "\\.(pem|key|pfx|p12|jks)$|^\\.env$"; exit 1\n'
      printf '          fi\n'
    } > .github/workflows/ci.yml
}
```

```bash
# --- guard install: prepare-commit-msg appends Keel-Version -----------------------------------
tv="$(fixture bare)"
( cd "$tv" && "$KEEL" init -y >/dev/null 2>&1 )
( cd "$tv" && "$KEEL" guard install >/dev/null 2>&1 )
[ -x "$tv/.githooks/prepare-commit-msg" ] && ok "guard install writes an executable prepare-commit-msg hook" \
  || bad "guard" "no executable .githooks/prepare-commit-msg"

( cd "$tv" && git add -A && git commit -q -m "a real commit" )
msg="$( cd "$tv" && git log -1 --format=%B )"
kv="$(sed -n 's/.*"keel_version": *"\([^"]*\)".*/\1/p' "$tv/.keel/profile.json" | head -1)"
case "$msg" in
  *"Keel-Version: $kv"*) ok "commit message carries the Keel-Version trailer" ;;
  *) bad "guard" "commit message has no 'Keel-Version: $kv' trailer. Got: $msg" ;;
esac

# amending does not duplicate the trailer. --no-edit, not -m: with -m git hands the hook a fresh
# message that has no trailer in it yet, so the duplicate guard is never reached and the assertion
# would pass against a hook with no guard at all. --no-edit feeds the previous message, trailer
# included, back through prepare-commit-msg, which is the case the guard exists for.
( cd "$tv" && git commit -q --amend --no-edit )
msg2="$( cd "$tv" && git log -1 --format=%B )"
count="$(printf '%s\n' "$msg2" | grep -c '^Keel-Version:')"

#!/usr/bin/env bash
# prepare-commit-msg hook, installed by `keel guard install`.
# Remove with `keel guard uninstall`.
set -uo pipefail

msg_file="${1:-}"
[ -n "$msg_file" ] && [ -f "$msg_file" ] || exit 0

HOOK
}

# Dispatches to the real checker. --json captures its human-readable stdout (the FAIL/WARN/ok
# lines already used throughout, and nothing else prints to fd 1 there) and reshapes it into one

        return $?

import json, re, sys

findings = []
problems = warnings = 0
for line in text.splitlines():

    "keel_version": keel_version or None,

    "verify_test_null": verify_test == "",

    "warnings": warnings,

}))

#!/usr/bin/env bash
# Fans out `keel doctor --json --fast` across repositories and prints one row per repository.
# Usage: keel-fleet <repo-path>...
#        keel-fleet --file <path-list-file>   (one repository path per line)
set -uo pipefail

# The sibling bin/keel first: the documented login-shell install symlinks that one file alone, and
# a fleet script is what a login shell runs. Falling back to PATH covers a plugin install. Made
# absolute, because the loop below cds into each repository before calling it.
KEEL_BIN="$(cd "$(dirname "$0")" && pwd)/keel"
[ -x "$KEEL_BIN" ] || KEEL_BIN=keel

paths=()
if [ "${1:-}" = "--file" ]; then
    while IFS= read -r line; do
        [ -n "$line" ] && paths+=("$line")
    done < "${2:?usage: keel-fleet --file <path-list-file>}"
else
    paths=("$@")
fi

[ "${#paths[@]}" -gt 0 ] || { printf 'usage: keel-fleet <repo-path>...\n       keel-fleet --file <path-list-file>\n' >&2; exit 1; }

status=0
printf 'repo\tkeel_version\tschema_version\tharnesses\tverify_test_null\tproblems\twarnings\n'
for p in "${paths[@]}"; do
    if ! out="$(cd "$p" 2>/dev/null && "$KEEL_BIN" doctor --json --fast 2>/dev/null)" || [ -z "$out" ]; then
        printf '%s\t(unreadable)\t\t\t\t\t\n' "$p"
        status=1
        continue
    fi
    row="$(printf '%s' "$out" | python3 -c "
import json, sys
d = json.load(sys.stdin)
cells = [str(d.get(k)) for k in ('keel_version', 'schema_version')]
cells.append(','.join(d.get('harnesses') or []))
cells += [str(d.get(k)) for k in ('verify_test_null', 'problems', 'warnings')]
print('\t'.join(cells))
")"
    printf '%s\t%s\n' "$p" "$row"
    printf '%s' "$out" | python3 -c "import json,sys; sys.exit(0 if json.load(sys.stdin)['problems']==0 else 1)" || status=1
done
exit "$status"
```
